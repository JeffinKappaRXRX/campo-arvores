// CAMPO — receptor de fotos, Excel e KML no seu Google Drive.
// 1. Substitua SOMENTE o ID abaixo pelo ID da sua pasta privada.
// 2. Execute configurarCampo uma vez e autorize o seu próprio projeto.
// 3. Copie a CHAVE_INTEGRACAO do registro de execução e envie ao responsável pelo app.
// 4. Implantar > Nova implantação > Aplicativo da Web > Executar como: Eu
//    Quem tem acesso: Qualquer pessoa. Copie a URL terminada em /exec.
// Não compartilhe a pasta ou a chave com os inspetores.
const PASTA_CAMPO_ID = 'COLE_AQUI_O_ID_DA_PASTA';

function configurarCampo() {
  if (PASTA_CAMPO_ID === 'COLE_AQUI_O_ID_DA_PASTA') throw new Error('Preencha o ID da pasta antes de executar.');
  const pasta = DriveApp.getFolderById(PASTA_CAMPO_ID);
  const props = PropertiesService.getScriptProperties();
  props.setProperty('CAMPO_FOLDER_ID', pasta.getId());
  if (!props.getProperty('CAMPO_SECRET')) props.setProperty('CAMPO_SECRET', Utilities.getUuid()+Utilities.getUuid());
  console.log('PASTA: '+pasta.getName());
  console.log('CHAVE_INTEGRACAO: '+props.getProperty('CAMPO_SECRET'));
}
function doGet() {
  return respostaCampo({ok:false,error:'Use o aplicativo Campo para enviar. Esta página não mostra arquivos.'});
}
function respostaCampo(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON);
}
function doPost(e) {
  let lock;
  try {
    const p = JSON.parse(e.postData.contents), props = PropertiesService.getScriptProperties();
    const secret = props.getProperty('CAMPO_SECRET');
    if (!secret || p.token !== secret) return respostaCampo({ok:false,code:'AUTH',error:'Integração não autorizada.'});
    if (p.action === 'status') {
      const folder = DriveApp.getFolderById(props.getProperty('CAMPO_FOLDER_ID'));
      return respostaCampo({ok:true,folderName:folder.getName(),consolidation:true,consolidatedUrl:props.getProperty('CAMPO_CONSOLIDADO_ID')?'https://drive.google.com/file/d/'+props.getProperty('CAMPO_CONSOLIDADO_ID')+'/view':null});
    }
    if(p.action==='consolidate'){
      if(!/^[a-zA-Z0-9_-]{1,100}$/.test(p.recordId||'')||!/^\d{4}-\d{2}-\d{2}T[\d:.Z+-]+$/.test(p.version||''))throw Error('INVALID');
      lock=LockService.getScriptLock();if(!lock.tryLock(10000))return respostaCampo({ok:false,code:'BUSY'});
      const parent=DriveApp.getFolderById(props.getProperty('CAMPO_FOLDER_ID'));
      const folders=parent.getFoldersByName('Campo_'+p.recordId+'_'+p.version.replace(/[^0-9]/g,''));if(!folders.hasNext())throw Error('VERIFY');
      const row=lerCadastroCampo(folders.next());if(!row||row.id!==p.recordId||row.updatedAt!==p.version)throw Error('VERIFY');
      const result=consolidarCampo(parent,row,false);return respostaCampo({ok:true,consolidated:true,recordId:p.recordId,version:p.version,url:result.url,count:result.count});
    }
    if (p.action !== 'file' || !/^[a-zA-Z0-9_-]{1,100}$/.test(p.recordId||'') || !/^\d{4}-\d{2}-\d{2}T[\d:.Z+-]+$/.test(p.version||'')) throw new Error('INVALID');
    const allowed = /^(?:[a-zA-Z0-9_-]+_(?:original\.(?:jpg|png|webp|heic)|georreferenciada\.jpg)|inspecao\.(?:xlsx|kml)|metadados\.json)$/;
    if (!allowed.test(p.name||'') || typeof p.base64 !== 'string' || p.base64.length>28000000 || !/^[A-Za-z0-9+/]*={0,2}$/.test(p.base64)) throw new Error('INVALID');
    const data = Utilities.base64Decode(p.base64);
    if (!data.length || data.length>20*1024*1024) throw new Error('SIZE');
    const hash = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256,data).map(b=>('0'+((b+256)%256).toString(16)).slice(-2)).join('');
    const mimes={xlsx:'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',kml:'application/vnd.google-earth.kml+xml',json:'application/json',jpg:'image/jpeg',png:'image/png',webp:'image/webp',heic:'image/heic'};
    lock=LockService.getScriptLock();if (!lock.tryLock(10000)) return respostaCampo({ok:false,code:'BUSY',error:'Envio ocupado. Tente novamente.'});
    const parent=DriveApp.getFolderById(props.getProperty('CAMPO_FOLDER_ID'));
    // Uma subpasta por versão: uma edição não sobrescreve a inspeção anterior.
    const folderName='Campo_'+p.recordId+'_'+p.version.replace(/[^0-9]/g,'');
    const folders=parent.getFoldersByName(folderName), folder=folders.hasNext()?folders.next():parent.createFolder(folderName);
    const existing=folder.getFilesByName(p.name);let file;
    if (existing.hasNext()) {
      file=existing.next();
      // Conteúdo original permanece imutável em uma mesma versão.
      const old=Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256,file.getBlob().getBytes()).map(b=>('0'+((b+256)%256).toString(16)).slice(-2)).join('');
      if (old!==hash) return respostaCampo({ok:false,code:'CONFLICT',error:'Arquivo diferente com mesmo nome. Edite e salve a inspeção para criar nova versão.'});
    } else {
      file=folder.createFile(Utilities.newBlob(data,mimes[p.name.split('.').pop()],p.name));
    }
    if (file.getSize()!==data.length) throw new Error('VERIFY');
    // Metadados são enviados por último. Consolidar não substitui os arquivos individuais.
    if(p.name==='metadados.json'){try{const row=lerCadastroCampo(folder);if(row)consolidarCampo(parent,row,false);}catch(ignore){/* A ação consolidate retoma a atualização sem reenviar fotos. */}}
    return respostaCampo({ok:true,id:file.getId(),size:file.getSize(),sha256:hash,folderId:folder.getId(),folderUrl:folder.getUrl()});
  } catch(err) {
    const known={SIZE:'Arquivo acima de 20 MB.',INVALID:'Solicitação inválida.',VERIFY:'Google não confirmou o tamanho do arquivo.'};
    return respostaCampo({ok:false,code:known[err.message]?err.message:'GOOGLE',error:known[err.message]||'Falha no Google Drive. Confira permissões, espaço disponível e cotas no painel Execuções do Apps Script.'});
  } finally {if(lock&&lock.hasLock())lock.releaseLock();}
}

// Execute esta função uma vez ao atualizar o receptor: inclui os cadastros existentes.
function atualizarConsolidadoCampo(){
  const lock=LockService.getScriptLock();lock.waitLock(30000);
  try{const props=PropertiesService.getScriptProperties(),parent=DriveApp.getFolderById(props.getProperty('CAMPO_FOLDER_ID')||PASTA_CAMPO_ID);
    const result=consolidarCampo(parent,null,true);console.log('EXCEL_UNICO: '+result.url);console.log('CADASTROS: '+result.count);
  }finally{lock.releaseLock();}
}
function lerCadastroCampo(folder){
 const found=folder.getFilesByName('metadados.json');if(!found.hasNext())return null;
 const row=JSON.parse(found.next().getBlob().getDataAsString('UTF-8'));
 if(!/^[a-zA-Z0-9_-]{1,100}$/.test(row.id||'')||!Number.isFinite(Date.parse(row.updatedAt)))return null;
 for(const name of ['inspecao.xlsx','inspecao.kml'])if(!folder.getFilesByName(name).hasNext())return null;
 row.photos=(row.photos||[]).map(p=>{const out={...p};for(const [key,urlKey] of [['original','originalUrl'],['georreferenced','georreferencedUrl']]){
   if(p[key]){const it=folder.getFilesByName(p[key]);if(!it.hasNext())throw Error('VERIFY');out[urlKey]=it.next().getUrl();}
 }out.mime=p.original&&p.original.endsWith('.png')?'image/png':p.original&&p.original.endsWith('.webp')?'image/webp':p.original&&p.original.endsWith('.heic')?'image/heic':'image/jpeg';out.annotated=Boolean(p.georreferenced);return out;});
 row.folderUrl=folder.getUrl();row.consolidatedAt=new Date().toISOString();return row;
}
